cmake_minimum_required(VERSION 3.21)
project(asgard-home LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Gui Qml Quick)
add_executable(asgard-home main.cpp backend.cpp)
target_link_libraries(asgard-home PRIVATE Qt6::Gui Qt6::Qml Qt6::Quick)
install(TARGETS asgard-home DESTINATION bin)
install(DIRECTORY qml/ DESTINATION share/asgard-home/qml)
